cmake_minimum_required(VERSION 3.20)
project(fsot_chua_error_budget CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)
add_executable(eb src/eb.cpp)
target_compile_options(eb PRIVATE -O3 -march=native -ffp-contract=off -Wall -Wextra)
target_link_libraries(eb PRIVATE Threads::Threads)
add_executable(mc src/mc.cpp)
target_compile_options(mc PRIVATE -O3 -ffp-contract=off)
target_link_libraries(mc PRIVATE Threads::Threads)
